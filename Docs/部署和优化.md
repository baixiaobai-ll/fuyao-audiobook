### 2. 网络优化

#### 请求重试机制
```swift
func synthesizeWithRetry(
    segment: TextSegment,
    voice: Voice,
    maxRetries: Int = 3
) async throws -> AudioData {
    var lastError: Error?

    for attempt in 1...maxRetries {
        do {
            return try await ttsEngine.synthesize(segment: segment, voice: voice)
        } catch {
            lastError = error
            print("⚠️ 第 \(attempt) 次尝试失败，重试中...")
            try await Task.sleep(nanoseconds: UInt64(attempt) * 1_000_000_000)
        }
    }

    throw lastError ?? TTSError.networkError
}
```

## 错误处理和日志

### 1. 统一错误处理

```swift
enum AppError: LocalizedError {
    case analysisError(Error)
    case ttsError(Error)
    case mixerError(Error)
    case playerError(Error)
    case networkError(Error)
    case cacheError(Error)

    var errorDescription: String? {
        switch self {
        case .analysisError(let error):
            return "文本分析失败: \(error.localizedDescription)"
        case .ttsError(let error):
            return "语音合成失败: \(error.localizedDescription)"
        case .mixerError(let error):
            return "音频混合失败: \(error.localizedDescription)"
        case .playerError(let error):
            return "播放失败: \(error.localizedDescription)"
        case .networkError(let error):
            return "网络错误: \(error.localizedDescription)"
        case .cacheError(let error):
            return "缓存错误: \(error.localizedDescription)"
        }
    }

    var recoverySuggestion: String? {
        switch self {
        case .networkError:
            return "请检查网络连接后重试"
        case .ttsError:
            return "请检查 TTS API 配置"
        case .analysisError:
            return "请检查 AI API 配置"
        default:
            return "请稍后重试"
        }
    }
}
```

## 监控和分析

### 1. 性能监控

```swift
class PerformanceMonitor {
    static func measure<T>(_ operation: String, block: () throws -> T) rethrows -> T {
        let start = Date()
        defer {
            let duration = Date().timeIntervalSince(start)
            Logger.log("[\(operation)] 耗时: \(String(format: "%.2f", duration))秒")
        }
        return try block()
    }

    static func measureAsync<T>(
        _ operation: String,
        block: () async throws -> T
    ) async rethrows -> T {
        let start = Date()
        defer {
            let duration = Date().timeIntervalSince(start)
            Logger.log("[\(operation)] 耗时: \(String(format: "%.2f", duration))秒")
        }
        return try await block()
    }
}

// 使用
let result = await PerformanceMonitor.measureAsync("文本分析") {
    try await textAnalyzer.analyze(text: text)
}
```
